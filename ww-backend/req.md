-- Table: public.users

-- DROP TABLE IF EXISTS public.users;

CREATE TABLE IF NOT EXISTS public.users
(
    id integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    username character varying(50) COLLATE pg_catalog."default" NOT NULL,
    password character varying(255) COLLATE pg_catalog."default" NOT NULL,
    fullname character varying(100) COLLATE pg_catalog."default" NOT NULL,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    group_id integer,
    branch_id integer,
    dept_id integer,
    force_logout smallint DEFAULT 0,
    last_activity timestamp without time zone,
    expires_at timestamp without time zone,
    sync_lock smallint NOT NULL DEFAULT 0,
    avatar character varying(255) COLLATE pg_catalog."default",
    CONSTRAINT users_pkey PRIMARY KEY (id),
    CONSTRAINT users_username_key UNIQUE (username),
    CONSTRAINT fk_users_branch FOREIGN KEY (branch_id)
        REFERENCES public.branches (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE SET NULL,
    CONSTRAINT fk_users_dept FOREIGN KEY (dept_id)
        REFERENCES public.departments (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE SET NULL,
    CONSTRAINT fk_users_group FOREIGN KEY (group_id)
        REFERENCES public.user_groups (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE SET NULL
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.users
    OWNER to postgres;

-- Table: public.branches

-- DROP TABLE IF EXISTS public.branches;

CREATE TABLE IF NOT EXISTS public.branches
(
    id integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    branch_code character varying(20) COLLATE pg_catalog."default" NOT NULL,
    branch_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    api_url character varying(255) COLLATE pg_catalog."default",
    address text COLLATE pg_catalog."default",
    contact_number character varying(20) COLLATE pg_catalog."default",
    is_active boolean DEFAULT true,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT branches_pkey PRIMARY KEY (id)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.branches
    OWNER to postgres;

-- Table: public.user_groups

-- DROP TABLE IF EXISTS public.user_groups;

CREATE TABLE IF NOT EXISTS public.user_groups
(
    id integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    group_name character varying(50) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT user_groups_pkey PRIMARY KEY (id)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.user_groups
    OWNER to postgres;

-- Table: public.user_branches

-- DROP TABLE IF EXISTS public.user_branches;

CREATE TABLE IF NOT EXISTS public.user_branches
(
    user_id integer NOT NULL,
    branch_id integer NOT NULL,
    CONSTRAINT user_branches_pkey PRIMARY KEY (user_id, branch_id),
    CONSTRAINT fk_user_branches_branch FOREIGN KEY (branch_id)
        REFERENCES public.branches (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT fk_user_branches_user FOREIGN KEY (user_id)
        REFERENCES public.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.user_branches
    OWNER to postgres;

-- Table: public.group_permissions

-- DROP TABLE IF EXISTS public.group_permissions;

CREATE TABLE IF NOT EXISTS public.group_permissions
(
    id integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    group_id integer NOT NULL,
    menu_id integer NOT NULL,
    can_view boolean DEFAULT false,
    can_rpt boolean DEFAULT false,
    can_add boolean DEFAULT false,
    can_edit boolean DEFAULT false,
    can_delete boolean DEFAULT false,
    access_level smallint DEFAULT 3,
    CONSTRAINT group_permissions_pkey PRIMARY KEY (id),
    CONSTRAINT unique_permission UNIQUE (group_id, menu_id),
    CONSTRAINT fk_group_permissions_group FOREIGN KEY (group_id)
        REFERENCES public.user_groups (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT fk_group_permissions_menu FOREIGN KEY (menu_id)
        REFERENCES public.menus (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.group_permissions
    OWNER to postgres;

2. table postgresql => users ,branches ,user_groups ,user_branches ,group_permissions
- table : users
	MSS code -> PG username
	MSS นำ code มาเข้ารหัส ใช้เป็นรหัสผ่าน -> PG password
	MSS name -> PG fullname
	MSS group_name -> PG group_id -> มาหาใน table user_groups แล้วนำ ID ไปบันทึก
	MSS branch_code -> PG branch_id -> มาหาใน branches.branch_code แล้วนำ ID ไปบันทึก
	PG force_logout = 0
	PG sync_lock = 0
	column ไหนที่ไม่ได้ระบุ ใส่เป็นค่าเริ่มต้น
- table : user_branches
	MSS : site_manage เป็นรูปแบบข้อความเป็น array ตัวอย่าง ["WN","WT","WP","WO"] นำมา loop หาใน PG branches.branch_code แล้ว นำ ID ไปบันทึกใน user_branches
3. การ sync ข้อมูลรอบถัดไป ถ้า Map กับ ระหว่าง MSS กับ PG ถ้าไม่พบข้อมูลใน MSS ให้ Update table users.force_logout = 1 ปิดสิทธิการใช้งานระบบ
